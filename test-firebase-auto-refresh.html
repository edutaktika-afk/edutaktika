⚠️ How It Works:

// Firebase auto-refresh functionality async function loadFirebaseDesigns() { const db = firebase.database(); const designsRef = db.ref('designs'); // Listen for real-time updates designsRef.on('value', (snapshot) => { const designs = []; snapshot.forEach((child) => { const data = child.val(); if (data && data.json) { designs.push({ id: child.key, name: data.name || child.key, jsonUrl: `https://edutaktika-default-rtdb.firebaseio.com/designs/${child.key}.json`, thumbUrl: data.thumbnail || 'fallback-svg', created: data.createdAt ? new Date(data.createdAt).toLocaleDateString() : 'Unknown' }); } }); renderDesigns(designs, container); }); }
